MA5:=MA(C,5);
DIF:=EMA(C,12)-EMA(C,26);
DEA:=EMA(DIF,9);
MC:=CROSS(DIF,DEA);
MTM:=C-REF(C,12);
MX:=CROSS(MTM,0);
T1:=CROSS(MA5,MA(C,35)) OR CROSS(MA5,MA(C,28));
T2:=BARSLAST(MC)>=2 AND BARSLAST(MC)<=5;
T3:=BARSLAST(MX)>=2 AND BARSLAST(MX)<=5;
T4:=DIF>DEA AND DIF>0 AND MTM>0;
XG:T1 AND T2 AND T3 AND T4;
