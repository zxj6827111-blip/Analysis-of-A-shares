{先跌后涨5日内筛选指标_完整版}
MA5:=MA(C,5);
MA7:=MA(C,7);
MA28:=MA(C,28);
MA35:=MA(C,35);

MA5X28:=BARSLAST(CROSS(MA5,MA28))<=5;
MA7X35:=BARSLAST(CROSS(MA7,MA35))<=5;

DIF:=EMA(C,12)-EMA(C,26);
DEA:=EMA(DIF,9);
MACD:=2*(DIF-DEA);

DIF60:="MACD.DIF#MIN60";
DEA60:="MACD.DEA#MIN60";

M60_DS:=COUNT(CROSS(DEA60,DIF60),20)>=1;
M60_GS:=COUNT(CROSS(DIF60,DEA60),20)>=1;
M60_DTG:=M60_DS AND M60_GS AND BARSLAST(CROSS(DIF60,DEA60))<BARSLAST(CROSS(DEA60,DIF60));

MTM:=CLOSE-REF(CLOSE,12);
MTM_S:=BARSLAST(CROSS(0,MTM));
MTM_G:=BARSLAST(CROSS(MTM,0));
MTM_DTG:=MTM_S<=35 AND MTM_G<=35 AND MTM_G<MTM_S;

HIST_TURN:=REF(MACD,1)<=0 AND MACD>0;

ABOVE_MA:=C>MA28 AND C>MA35;
BULL_MA:=MA5>MA7 AND MA7>MA28 AND MA28>MA35;

XG:MA5X28 AND MA7X35 AND M60_DTG AND MTM_DTG AND HIST_TURN AND ABOVE_MA AND BULL_MA;
