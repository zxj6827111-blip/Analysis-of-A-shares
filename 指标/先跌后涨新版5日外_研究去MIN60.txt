MA5:=MA(C,5);
MA7:=MA(C,7);
MA28:=MA(C,28);
MA35:=MA(C,35);

MA5X28:=BARSLAST(CROSS(MA5,MA28))>5;
MA7X35:=BARSLAST(CROSS(MA7,MA35))>5;

DIF:=EMA(C,12)-EMA(C,26);
DEA:=EMA(DIF,9);
MACD:=2*(DIF-DEA);

MTM:=CLOSE-REF(CLOSE,12);
MTM_S:=BARSLAST(CROSS(0,MTM));
MTM_G:=BARSLAST(CROSS(MTM,0));
MTM_DTG:=MTM_S<=14 AND MTM_G<=14 AND MTM_G<MTM_S;

HIST_TURN:=REF(MACD,1)<=0 AND MACD>0;

ABOVE_MA:=C>MA28 OR C>MA35;

XG:MA5X28 AND MA7X35 AND MTM_DTG AND HIST_TURN AND ABOVE_MA;
